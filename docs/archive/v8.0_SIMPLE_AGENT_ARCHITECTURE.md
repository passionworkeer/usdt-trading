# src/ai/nlt_translator.py

class NLTDataTranslator:
    """自然语言数据翻译器"""

    def translate_micro_snapshot(self, snapshot: Dict) -> str:
        """
        将微观数据翻译成技术面报告

        Input: {
            "symbol": "BTCUSDT",
            "price": 95000,
            "funding_rate": 0.0008,
            "obi": -0.6,
            "volume_15m": 1200000,
            "ema_distance": 0.02
        }

        Output:
        "技术面报告：当前资金费率 0.08%（处于历史极高拥挤区，多头过度贪婪）。
         盘口失衡率 -0.6，上方存在冰山卖单压制。15分钟级别刚刚放量突破 EMA50，
         成交量放大至 120 万 USDT。价格偏离 EMA50 仅 2%，处于健康击球区。"
        """
        sections = []

        # 1. 资金费率解读
        funding = snapshot.get('funding_rate', 0) * 100
        if funding > 0.05:
            sections.append(
                f"资金费率 {funding:.2f}%（历史极高拥挤区，多头过度贪婪）"
            )
        elif funding > 0.01:
            sections.append(
                f"资金费率 {funding:.2f}%（温和多头情绪）"
            )
        elif funding < -0.05:
            sections.append(
                f"资金费率 {funding:.2f}%（极度恐慌，空头拥挤）"
            )
        else:
            sections.append(
                f"资金费率 {funding:.2f}%（中性区域）"
            )

        # 2. 订单簿失衡
        obi = snapshot.get('obi', 0)
        if obi > 0.5:
            sections.append(
                f"盘口失衡率 +{obi:.1f}，下方冰山买单支撑强劲"
            )
        elif obi < -0.5:
            sections.append(
                f"盘口失衡率 {obi:.1f}，上方冰山卖单压制明显"
            )
        else:
            sections.append(
                f"盘口失衡率 {obi:.1f}，买卖力量均衡"
            )

        # 3. 成交量分析
        volume = snapshot.get('volume_15m', 0)
        avg_volume = snapshot.get('avg_volume_15m', volume)
        if volume > avg_volume * 2:
            sections.append(
                f"15分钟级别巨量爆发，成交量达 {volume/1e6:.1f}M USDT " +
                f"（平均 {avg_volume/1e6:.1f}M），主力资金进场信号"
            )
        elif volume > avg_volume * 1.5:
            sections.append(
                f"15分钟级别放量突破，成交量 {volume/1e6:.1f}M USDT"
            )

        # 4. EMA 距离
        ema_dist = snapshot.get('ema_distance', 0) * 100
        if abs(ema_dist) < 3:
            sections.append(
                f"价格紧贴 EMA50（偏离 {ema_dist:+.1f}%），健康回踩击球区"
            )
        elif ema_dist > 5:
            sections.append(
                f"价格大幅偏离 EMA50（+{ema_dist:.1f}%），追高风险加大"
            )

        return "技术面报告：" + "；".join(sections) + "。"
```

# src/intelligence/web_scraper.py

import requests
from bs4 import BeautifulSoup
from typing import List, Dict
import logging

logger = logging.getLogger(__name__)

class IntelligenceSniffer:
    """外部情报嗅探器"""

    def __init__(self, twitter_bearer_token: str = None):
        self.twitter_token = twitter_bearer_token

    async def scrape_twitter_sentiment(
        self,
        symbol: str,
        limit: int = 50
    ) -> List[Dict]:
        """
        抓取 Twitter/X 上特定币种的最新推文

        注意：需要 Twitter API v2 Bearer Token
        如果没有 token，返回空列表（不影响交易）
        """
        if not self.twitter_token:
            logger.warning("未配置 Twitter API Token，跳过情绪抓取")
            return []

        try:
            # 提取币种名称（BTCUSDT -> Bitcoin）
            coin_name = self._symbol_to_name(symbol)

            # Twitter API v2 搜索
            url = "https://api.twitter.com/2/tweets/search/recent"
            headers = {"Authorization": f"Bearer {self.twitter_token}"}
            params = {
                "query": f"{coin_name} -is:retweet lang:en",
                "max_results": min(limit, 100),
                "tweet.fields": "created_at,public_metrics"
            }

            response = requests.get(
                url,
                headers=headers,
                params=params,
                timeout=10
            )

            if response.status_code != 200:
                logger.error(f"Twitter API 错误: {response.status_code}")
                return []

            data = response.json()
            tweets = []

            for tweet in data.get('data', []):
                tweets.append({
                    'text': tweet['text'],
                    'likes': tweet['public_metrics']['like_count'],
                    'retweets': tweet['public_metrics']['retweet_count'],
                    'created_at': tweet['created_at']
                })

            logger.info(f"抓取到 {len(tweets)} 条推文")
            return tweets

        except Exception as e:
            logger.error(f"Twitter 抓取失败: {e}")
            return []

    async def scrape_macro_news(self, limit: int = 10) -> List[Dict]:
        """
        抓取过去 24 小时的宏观新闻

        来源：CoinDesk, Bloomberg Crypto
        """
        news = []

        # CoinDesk Crypto
        try:
            response = requests.get(
                "https://www.coindesk.com/arc/outboundfeeds/rss/",
                timeout=10
            )
            soup = BeautifulSoup(response.content, 'xml')
            items = soup.find_all('item')[:limit]

            for item in items:
                news.append({
                    'title': item.title.text,
                    'source': 'CoinDesk',
                    'published': item.pubDate.text
                })
        except Exception as e:
            logger.error(f"CoinDesk 抓取失败: {e}")

        logger.info(f"抓取到 {len(news)} 条新闻")
        return news

    def _symbol_to_name(self, symbol: str) -> str:
        """BTCUSDT -> Bitcoin"""
        mapping = {
            'BTC': 'Bitcoin',
            'ETH': 'Ethereum',
            'SOL': 'Solana',
            'BNB': 'Binance'
        }
        base = symbol.replace('USDT', '').replace('BUSD', '')
        return mapping.get(base, base)
```

class MacroOracle:
    """宏观大局观生成器"""

    def __init__(self, decision_engine: ClaudeDecisionEngine):
        self.engine = decision_engine
        self.last_macro_state = None
        self.last_update_time = None

    async def generate_macro_state(
        self,
        intelligence: Dict
    ) -> Dict:
        """
        生成宏观大局观（每小时调用一次）

        Input: {
            'tweets': [...],
            'news': [...]
        }

        Output: {
            'global_sentiment': 'panic' | 'neutral' | 'euphoric',
            'dominant_narrative': 'DeFi复苏' | '监管恐慌' | ...,
            'trading_bans': ['LONG'],  # 禁止做多
            'recommended_stance': 'defensive'
        }
        """
        # 构建 prompt
        prompt = self._build_macro_prompt(intelligence)

        # 调用 Claude
        response = self.engine.client.messages.create(
            model="claude-sonnet-4-5-20250514",
            max_tokens=1000,
            messages=[{"role": "user", "content": prompt}]
        )

        # 解析响应
        content = response.content[0].text
        result = self._parse_response(content)

        # 缓存
        self.last_macro_state = result
        self.last_update_time = datetime.now()

        logger.info(f"宏观大局观更新: {result['global_sentiment']}")
        return result

    def _build_macro_prompt(self, intelligence: Dict) -> str:
        """构建宏观分析 prompt"""

        tweets = intelligence.get('tweets', [])
        news = intelligence.get('news', [])

        # 提取推文样本
        tweet_sample = "\n".join([
            f"- {t['text'][:100]}... ({t['likes']} likes)"
            for t in tweets[:10]
        ])

        # 提取新闻标题
        news_sample = "\n".join([
            f"- {n['title']} ({n['source']})"
            for n in news[:10]
        ])

## Twitter 情绪（过去 1 小时）
{tweet_sample if tweet_sample else "无数据"}

## 宏观新闻（过去 24 小时）
{news_sample if news_sample else "无数据"}

## 你的任务
基于以上信息，生成一个宏观大局观报告，格式如下：

```json

### Phase 4: Slippage Hardlock (30分钟)

**目标**: 5秒滑点硬拦截

**实现**:
```python
# src/execution/slippage_hardlock.py

import time
from typing import Tuple

class SlippageHardlock:
    """滑点硬拦截器"""

    def __init__(self, threshold_pct: float = 0.5, timeout_sec: float = 5.0):
        self.threshold_pct = threshold_pct  # 0.5%
        self.timeout_sec = timeout_sec      # 5 秒

        self._locked_prices = {}  # {symbol: (price, timestamp)}

    def lock_trigger_price(self, symbol: str, price: float) -> None:
        """锁定触发价格"""
        self._locked_prices[symbol] = (price, time.time())
        logger.info(f"[{symbol}] 锁定触发价格: ${price:,.2f}")

    def check_slippage(self, symbol: str, current_price: float) -> Tuple[bool, str]:
        """
        检查滑点是否超过阈值

        Returns:
            (是否通过, 原因)
        """
        if symbol not in self._locked_prices:
            return False, "未锁定价格"

        locked_price, timestamp = self._locked_prices[symbol]

        # 检查是否超时
        elapsed = time.time() - timestamp
        if elapsed > self.timeout_sec:
            return False, f"决策超时（{elapsed:.1f}s > {self.timeout_sec}s）"

        # 计算滑点
        slippage_pct = abs(current_price - locked_price) / locked_price * 100

        if slippage_pct > self.threshold_pct:
            return False, (
                f"滑点过大（{slippage_pct:.2f}% > {self.threshold_pct}%），"
                f"触发价 ${locked_price:,.2f} → 当前价 ${current_price:,.2f}"
            )

        logger.info(
            f"[{symbol}] 滑点检查通过: {slippage_pct:.2f}% "
            f"({elapsed:.1f}s)"
        )
        return True, "滑点在可接受范围内"
```

from src.ai.nlt_translator import NLTDataTranslator
from src.ai.macro_oracle import MacroOracle
from src.intelligence.web_scraper import IntelligenceSniffer
from src.execution.slippage_hardlock import SlippageHardlock

class SniperTrader:
    def __init__(self):
        # ... 现有代码 ...

        # v8.0 新组件
        self.translator = NLTDataTranslator()
        self.scraper = IntelligenceSniffer(
            twitter_bearer_token=os.getenv('TWITTER_BEARER_TOKEN')
        )
        self.macro_oracle = MacroOracle(self.decision_engine)
        self.slippage_guard = SlippageHardlock(
            threshold_pct=0.5,
            timeout_sec=5.0
        )

        # 宏观状态（每小时更新）
        self.macro_state = None
        self.last_macro_update = None

    async def _update_macro_state(self):
        """每小时更新宏观大局观"""
        # 抓取外部情报
        intelligence = {
            'tweets': await self.scraper.scrape_twitter_sentiment(
                self.symbol, limit=50
            ),
            'news': await self.scraper.scrape_macro_news(limit=10)
        }

        # 生成宏观状态
        self.macro_state = await self.macro_oracle.generate_macro_state(
            intelligence
        )
        self.last_macro_update = datetime.now()

    async def _on_mtf_trigger(self, mtf_signal: Dict):
        """MTF 放量突破触发"""
        # 1. 锁定价格
        current_price = mtf_signal['price']
        self.slippage_guard.lock_trigger_price(self.symbol, current_price)

        # 2. 翻译微观数据
        micro_report = self.translator.translate_micro_snapshot(
            {
                'symbol': self.symbol,
                'price': current_price,
                'funding_rate': self.funding_rate,
                'obi': self.order_book_imbalance,
                'volume_15m': self.volume_15m,
                'ema_distance': (current_price - self.ema50) / self.ema50
            }
        )

        # 3. 请求 AI 确认
        decision = await self.decision_engine.analyze_market(
            symbol=self.symbol,
            price=current_price,
            price_history=self.price_history,
            market_data={
                'micro_report': micro_report,
                'macro_state': self.macro_state,
                'mtf_signal': mtf_signal
            }
        )

        # 4. 滑点检查
        latest_price = await self.get_current_price()
        passed, reason = self.slippage_guard.check_slippage(
            self.symbol, latest_price
        )

        if not passed:
            logger.warning(f"滑点拦截: {reason}")
            return

        # 5. 执行交易
        if decision.action != 'hold':
            await self.execute_trade(decision)
```
